        """Shared in-memory rate limiter. Factored from public_booking.py (10.2)."""
        from __future__ import annotations
        import time
        from collections import defaultdict, deque
        from fastapi import HTTPException, status

        _store: dict[str, deque[float]] = defaultdict(deque)
        DEFAULT_WINDOW_SECONDS = 60
        DEFAULT_MAX_REQUESTS = 10

        def check_rate_limit(ip: str, window_seconds: int = DEFAULT_WINDOW_SECONDS, max_requests: int = DEFAULT_MAX_REQUESTS) -> None:
            now = time.monotonic()
            q = _store[ip]
            # purge old
            while q and q[0] < now - window_seconds:
                q.popleft()
            if len(q) >= max_requests:
                raise HTTPException(status_code=status.HTTP_429_TOO_MANY_REQUESTS, detail="rate_limited")
            q.append(now)

        from __future__ import annotations
        import time
        from datetime import datetime, timezone
        import os
        import httpx
        from fastapi import APIRouter, Depends, Request
        from sqlalchemy import text
        from sqlalchemy.ext.asyncio import AsyncSession
        from backend.core.config import settings
        from backend.core.rate_limit import check_rate_limit
        from backend.db.session import get_db
        from backend.db.models.public.saas import SystemHealthSample
        from backend.schemas.system import HealthResponse, DependencyStatus

        router = APIRouter(prefix="/api/system", tags=["system"])

        async def _probe_postgres(db: AsyncSession) -> tuple[str, int]:
            start = time.perf_counter()
            try:
                await db.execute(text("SELECT 1"))
                status = "ok"
            except Exception:
                status = "down"
            return status, int((time.perf_counter() - start) * 1000)

        async def _probe_supabase() -> tuple[str, int]:
            start = time.perf_counter()
            try:
                async with httpx.AsyncClient(timeout=2.0) as c:
                    r = await c.get(f"{settings.SUPABASE_URL}/.well-known/openid-configuration")
                    status = "ok" if r.status_code == 200 else "degraded"
            except Exception:
                status = "down"
            return status, int((time.perf_counter() - start) * 1000)

        async def sample_health_now(db: AsyncSession) -> HealthResponse:
            pg_status, pg_ms = await _probe_postgres(db)
            auth_status, auth_ms = await _probe_supabase()
            all_green = pg_status == "ok" and auth_status == "ok"
            sample = SystemHealthSample(
                api_status="ok",
                pg_status=pg_status,
                pg_latency_ms=pg_ms,
                auth_status=auth_status,
                auth_latency_ms=auth_ms,
                all_green=all_green,
            )
            db.add(sample)
            await db.commit()
            version = (os.getenv("GIT_SHA") or "unknown")[:7]
            return HealthResponse(
                api="ok",
                postgres=DependencyStatus(status=pg_status, latency_ms=pg_ms),
                supabase_auth=DependencyStatus(status=auth_status, latency_ms=auth_ms),
                version=version,
                checked_at=datetime.now(timezone.utc).isoformat().replace("+00:00", "Z"),
            )

        @router.get("/health/", response_model=HealthResponse)
        async def get_health(request: Request, db: AsyncSession = Depends(get_db)):
            ip = (request.client.host if request.client else "unknown")
            check_rate_limit(ip, window_seconds=60, max_requests=10)
            return await sample_health_now(db)

        import asyncio
        import os
        from backend.api.routes.system import sample_health_now
        from backend.db.session import get_db_context  # if no helper, use a fresh session inline

        _pinger_task: asyncio.Task | None = None

        async def _health_pinger_loop():
            from backend.db.session import AsyncSessionLocal  # local import to avoid circular
            while True:
                try:
                    async with AsyncSessionLocal() as db:
                        await sample_health_now(db)
                except Exception:
                    pass  # never crash the loop; real exceptions are captured by Sentry
                await asyncio.sleep(60)

        @app.on_event("startup")
        async def _start_health_pinger():
            if os.getenv("SENTRY_ENVIRONMENT") != "production":
                return
            global _pinger_task
            _pinger_task = asyncio.create_task(_health_pinger_loop())

        @app.on_event("shutdown")
        async def _stop_health_pinger():
            global _pinger_task
            if _pinger_task:
                _pinger_task.cancel()
                try:
                    await _pinger_task
                except (asyncio.CancelledError, Exception):
