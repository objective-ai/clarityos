    2. Create components/admin/SystemStatusSection.tsx (the composer):

        "use client";
        import { useCallback, useEffect, useState } from "react";
        import { ServiceHealthPanel } from "@/components/admin/system/ServiceHealthPanel";
        import { RecentErrorsPanel } from "@/components/admin/system/RecentErrorsPanel";
        import { UptimePanel } from "@/components/admin/system/UptimePanel";

        const POLL_MS = 30_000;

        export function SystemStatusSection() {
          const [refreshKey, setRefreshKey] = useState(0);
          const [lastUpdated, setLastUpdated] = useState<Date | null>(null);

          const onRefresh = useCallback(() => {
            setRefreshKey(k => k + 1);
            setLastUpdated(new Date());
          }, []);

          useEffect(() => {
            // Initial mark + scheduled refresh
            setLastUpdated(new Date());
            const id = setInterval(onRefresh, POLL_MS);
            return () => clearInterval(id);
          }, [onRefresh]);

          return (
            <section className="space-y-6">
              <header className="flex items-center justify-between">
                <h1 className="text-2xl font-semibold">System Status</h1>
                <div className="flex items-center gap-3 text-sm text-white/60">
                  <span>Updated {lastUpdated ? lastUpdated.toLocaleTimeString([], { hour12: false }) : "--:--:--"}</span>
                  <button
                    type="button"
                    onClick={onRefresh}
                    className="px-3 py-1.5 rounded-md border border-white/10 hover:border-white/30 text-white"
                  >
                    Refresh
                  </button>
                </div>
              </header>
              <ServiceHealthPanel refreshKey={refreshKey} onFetched={() => setLastUpdated(new Date())} />
              <RecentErrorsPanel refreshKey={refreshKey} />
              <UptimePanel refreshKey={refreshKey} />
            </section>
          );
        }

    3. Create components/admin/system/ServiceHealthPanel.tsx:

        "use client";
        import { useEffect, useState } from "react";
        import type { HealthResponse } from "@/types/system";

        const dotColor = (status: string) =>
          status === "ok" ? "bg-[#2DD4BF]" : status === "degraded" ? "bg-amber-400" : "bg-red-500";

        export function ServiceHealthPanel({ refreshKey, onFetched }: { refreshKey: number; onFetched?: () => void }) {
          const [data, setData] = useState<HealthResponse | null>(null);
          const [error, setError] = useState<string | null>(null);

          useEffect(() => {
            let ok = true;
            fetch("/api/system/health")
              .then(r => r.ok ? r.json() : Promise.reject(new Error(String(r.status))))
              .then((d: HealthResponse) => { if (ok) { setData(d); setError(null); onFetched?.(); } })
              .catch(e => { if (ok) setError(String(e)); });
            return () => { ok = false; };
          }, [refreshKey, onFetched]);

          return (
            <div className="glass-card p-5">
              <h2 className="text-lg font-medium mb-4">Service Health</h2>
              {error && <div className="text-red-400 text-sm">Health probe failed: {error}</div>}
              {data && (
                <div className="grid grid-cols-1 md:grid-cols-3 gap-4">
                  <HealthCard name="FastAPI" status={data.api} latencyMs={null} />
                  <HealthCard name="Postgres" status={data.postgres.status} latencyMs={data.postgres.latencyMs} />
                  <HealthCard name="Supabase Auth" status={data.supabaseAuth.status} latencyMs={data.supabaseAuth.latencyMs} />
                </div>
              )}
            </div>
          );
        }

        function HealthCard({ name, status, latencyMs }: { name: string; status: string; latencyMs: number | null }) {
          return (
            <div className="border border-white/10 rounded-lg p-4">
              <div className="flex items-center gap-2">
                <span className={`inline-block w-2.5 h-2.5 rounded-full ${dotColor(status)}`} />
                <span className="font-medium">{name}</span>
              </div>
              <div className="mt-2 text-sm text-white/70">
                {status}{latencyMs !== null ? ` · ${latencyMs}ms` : ""}
              </div>
            </div>
          );
        }

       NOTE on camelCase: The FastAPI response uses snake_case; apiFetch recursively camelCases per project convention — so `supabase_auth` becomes `supabaseAuth` and `latency_ms` becomes `latencyMs` in the frontend. If the BFF proxy returns snake_case directly (no apiFetch wrapper), adjust the types — but the standard project pattern is camelCase on the client.

        "use client";
        import { useEffect, useState } from "react";
        import type { ErrorIssueList, ErrorIssue } from "@/types/system";

        export function RecentErrorsPanel({ refreshKey }: { refreshKey: number }) {
          const [data, setData] = useState<ErrorIssueList | null>(null);
          useEffect(() => {
            let ok = true;
            fetch("/api/system/errors")
              .then(r => r.json())
              .then((d: ErrorIssueList) => { if (ok) setData(d); })
              .catch(() => { if (ok) setData({ issues: [], fetchedAt: new Date().toISOString(), cached: false }); });
            return () => { ok = false; };
          }, [refreshKey]);

          return (
            <div className="glass-card p-5">
              <h2 className="text-lg font-medium mb-4">Recent Errors {data?.cached ? <span className="text-xs text-white/40">(cached)</span> : null}</h2>
              <div className="overflow-x-auto">
                <table className="w-full text-sm">
                  <thead>
                    <tr className="text-left text-white/50">
                      <th className="py-2">Title</th>
                      <th>Count</th>
                      <th>Last Seen</th>
                      <th>Env</th>
                    </tr>
                  </thead>
                  <tbody>
                    {(data?.issues ?? []).map((i: ErrorIssue) => (
                      <tr key={i.id} className="border-t border-white/5">
                        <td className="py-2">
                          <a href={i.permalink} target="_blank" rel="noreferrer" className="text-[#2DD4BF] hover:underline">
                            {i.title}
                          </a>
                        </td>
                        <td>{i.count}</td>
                        <td>{new Date(i.lastSeen).toLocaleString()}</td>
                        <td className="text-white/60">{i.environment ?? "—"}</td>
                      </tr>
                    ))}
                    {(!data || data.issues.length === 0) && (
                      <tr><td colSpan={4} className="py-4 text-white/50">No unresolved issues.</td></tr>
                    )}
                  </tbody>
                </table>
              </div>
            </div>
          );
        }

        function rollup(d: HealthResponse | null): DotColor {

          red: "bg-red-500",

        };
