        from datetime import timedelta
        from sqlalchemy import select, func as sa_func
        from backend.schemas.system import UptimeSummary

        @router.get("/uptime/", response_model=UptimeSummary)
        async def get_uptime(request: Request, db: AsyncSession = Depends(get_db)):
            ip = (request.client.host if request.client else "unknown")
            check_rate_limit(ip, window_seconds=60, max_requests=30)  # more lenient than /health
            window_start_cutoff = datetime.now(timezone.utc) - timedelta(days=7)
            q = select(
                sa_func.count(SystemHealthSample.id).label("total"),
                sa_func.count(sa_func.nullif(SystemHealthSample.all_green, False)).label("green"),
                sa_func.min(SystemHealthSample.checked_at).label("window_start"),
                sa_func.max(SystemHealthSample.checked_at).label("window_end"),
            ).where(SystemHealthSample.checked_at >= window_start_cutoff)
            result = (await db.execute(q)).one()
            total = result.total or 0
            green = result.green or 0
            pct = round((green / total) * 100.0, 2) if total > 0 else 0.0
            return UptimeSummary(
                uptime_pct=pct,
                samples_total=total,
                samples_green=green,
                window_start=result.window_start.isoformat().replace("+00:00", "Z") if result.window_start else None,
                window_end=result.window_end.isoformat().replace("+00:00", "Z") if result.window_end else None,
            )

        import { NextRequest, NextResponse } from "next/server";
        import { createClient } from "@/lib/supabase/server";  // adjust import to the project's actual server client helper
        import type { ErrorIssueList } from "@/types/system";

        let cache: { at: number; payload: ErrorIssueList } | null = null;
        const TTL_MS = 20_000;

        export async function GET(req: NextRequest) {
          // OWNER gate
          const supabase = createClient();
          const { data: { user } } = await supabase.auth.getUser();
          const role = (user?.app_metadata as any)?.role ?? (user as any)?.role;
          if (!user || role !== "OWNER") {
            return NextResponse.json({ error: "forbidden" }, { status: 403 });
          }

          const now = Date.now();
          if (cache && now - cache.at < TTL_MS) {
            return NextResponse.json({ ...cache.payload, cached: true });
          }

          const org = process.env.SENTRY_ORG;
          const project = process.env.SENTRY_PROJECT;
          const token = process.env.SENTRY_API_TOKEN;
          if (!org || !project || !token) {
            const payload: ErrorIssueList = { issues: [], fetchedAt: new Date().toISOString(), cached: false };
            return NextResponse.json(payload, { status: 200, headers: { "x-sentry-upstream-status": "not-configured" } });
          }

          const url = `https://sentry.io/api/0/projects/${org}/${project}/issues/?query=is:unresolved&limit=50&environment=production`;
          const r = await fetch(url, { headers: { Authorization: `Bearer ${token}` }, cache: "no-store" });
          if (!r.ok) {
            const payload: ErrorIssueList = { issues: [], fetchedAt: new Date().toISOString(), cached: false };
            return NextResponse.json(payload, { status: 200, headers: { "x-sentry-upstream-status": String(r.status) } });
          }
          const raw = await r.json() as Array<any>;
          const issues = raw.map(i => ({
            id: String(i.id),
            title: String(i.title ?? ""),
            count: Number(i.count ?? 0),
            userCount: Number(i.userCount ?? 0),
            lastSeen: String(i.lastSeen ?? ""),
            firstSeen: String(i.firstSeen ?? ""),
            permalink: String(i.permalink ?? ""),
            environment: i.environment ?? undefined,
            level: i.level ?? undefined,
            culprit: i.culprit ?? undefined,
          }));
          const payload: ErrorIssueList = { issues, fetchedAt: new Date().toISOString(), cached: false };
          cache = { at: now, payload };
          return NextResponse.json(payload);
        }
