       ```ts
       import { NextRequest } from "next/server";
       import { getAuthHeaders } from "@/lib/auth-headers";  // verify actual path — read inventoryStore donor

       export async function POST(
         req: NextRequest,
         { params }: { params: Promise<{ orderId: string }> }
       ) {
         const { orderId } = await params;
         const headers = await getAuthHeaders(req);
         const upstream = process.env.FASTAPI_URL || "http://localhost:8000";
         const resp = await fetch(`${upstream}/api/optical-orders/${orderId}/job-ticket/`, {
           method: "POST",
           headers,
         });
         if (!resp.ok) {
           const text = await resp.text();
           return new Response(text, { status: resp.status });
         }
         // Stream PDF body + Content-Disposition through unchanged
         const body = await resp.arrayBuffer();
         return new Response(body, {
           status: 200,
           headers: {
             "Content-Type": resp.headers.get("Content-Type") ?? "application/pdf",
             "Content-Disposition": resp.headers.get("Content-Disposition") ?? `attachment; filename="job-ticket-${orderId.slice(0,8)}.pdf"`,
           },
         });
       }
       ```
       Confirm the actual `getAuthHeaders` import path by checking `store/inventoryStore.ts` (STATE.md `[Phase 13-09]` confirms it's the raw-fetch donor). If the helper lives at `@/lib/auth` or similar, use that path.
