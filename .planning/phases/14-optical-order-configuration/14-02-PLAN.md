    ```python
    @router.get("/types/", response_model=list[LensTypeResponse])
    async def list_lens_types(
        ctx: TenantContext = Depends(resolve_tenant_context),
        db: AsyncSession = Depends(get_db_session),
        include_inactive: bool = False,
    ) -> list[LensType]:
        stmt = select(LensType).where(LensType.tenant_id == ctx.tenant_id)
        if not include_inactive:
            stmt = stmt.where(LensType.is_active.is_(True))
        stmt = stmt.order_by(LensType.display_order, LensType.name)
        result = await db.execute(stmt)
        return list(result.scalars().all())

    @router.post("/types/", response_model=LensTypeResponse, status_code=201,
                 dependencies=[Depends(require_clinical_action(ClinicalAction.MANAGE_LENS_CATALOG))])
    async def create_lens_type(
        payload: LensTypeCreate,
        ctx: TenantContext = Depends(resolve_tenant_context),
        db: AsyncSession = Depends(get_db_session),
    ) -> LensType:
        staff = await resolve_staff(ctx, db)
        row = LensType(tenant_id=ctx.tenant_id, **payload.model_dump())
        db.add(row)
        try:
            await db.flush()
        except IntegrityError as e:
            await db.rollback()
            raise HTTPException(409, detail={"error": "duplicate_name", "field": "name"}) from e
        await log_action(
            db, tenant_id=ctx.tenant_id, action=AuditAction.LENS_TYPE_CREATE,
            actor_id=staff.id if staff else None,
            target_type="lens_type", target_id=row.id, metadata={"name": row.name},
        )
        await db.commit()
        await db.execute(select(LensType).where(LensType.id == row.id))  # selectinload-friendly refetch
        return row

    @router.get("/types/{type_id}/", response_model=LensTypeResponse)
    async def get_lens_type(type_id: uuid.UUID, ctx=Depends(resolve_tenant_context), db=Depends(get_db_session)):
        row = await db.get(LensType, type_id)
        if not row or row.tenant_id != ctx.tenant_id:
            raise HTTPException(404, "Lens type not found")
        return row

    @router.patch("/types/{type_id}/", response_model=LensTypeResponse,
                  dependencies=[Depends(require_clinical_action(ClinicalAction.MANAGE_LENS_CATALOG))])
    async def update_lens_type(
        type_id: uuid.UUID, payload: LensTypeUpdate,
        ctx=Depends(resolve_tenant_context), db=Depends(get_db_session),
    ):
        row = await db.get(LensType, type_id)
        if not row or row.tenant_id != ctx.tenant_id:
            raise HTTPException(404)
        staff = await resolve_staff(ctx, db)
        for k, v in payload.model_dump(exclude_unset=True).items():
            setattr(row, k, v)
        try:
            await db.flush()
        except IntegrityError as e:
            await db.rollback()
            raise HTTPException(409, detail={"error": "duplicate_name", "field": "name"}) from e
        await log_action(
            db, tenant_id=ctx.tenant_id, action=AuditAction.LENS_TYPE_CREATE,  # reuse with metadata discriminator per §H
            actor_id=staff.id if staff else None, target_type="lens_type",
            target_id=row.id, metadata={"action": "update"},
        )
        await db.commit()
        return row
    ```
