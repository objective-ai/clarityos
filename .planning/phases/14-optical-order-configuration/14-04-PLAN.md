    ```python
    """Phase 14: Deterministic keyword scanner over saved AI Scribe output.

    Pure function. NO Claude calls. NO streaming. NO DB writes.
    Reads encounter.ai_summary_text + encounter.assessment_and_plan (both plain text per MEMORY.md / Pitfall 8).
    """
    from __future__ import annotations
    from typing import Any, Optional

    LENS_TYPE_KEYWORDS: dict[str, list[str]] = {
        "progressive": ["progressive", "PAL", "multifocal"],
        "bifocal": ["bifocal", "lined bifocal"],
        "single_vision": ["single vision", " SV "],  # leading/trailing spaces to avoid "SVD" etc.
        "reading": ["reading glasses", "near-only", "+2.00", "+2.25", "+2.50"],
    }

    MATERIAL_KEYWORDS: dict[str, list[str]] = {
        "polycarbonate": ["polycarbonate", "poly", "impact-resistant", "child", "pediatric"],
        "trivex": ["trivex"],
        "hi-index 1.67": ["hi-index 1.67", "1.67", "high-index"],
        "hi-index 1.74": ["hi-index 1.74", "1.74"],
        "hi-index 1.80": ["1.80"],
        "CR-39": ["CR-39", "standard plastic"],
    }

    COATING_KEYWORDS: dict[str, list[str]] = {
        "AR": ["anti-reflective", "AR coating", "no-glare", "anti-glare"],
        "blue light": ["blue light", "blue-blocker", "digital eye strain"],
        "photochromic": ["photochromic", "transitions", "transition lens"],
        "polarized": ["polarized"],
        "UV": ["UV protection", "ultraviolet"],
        "scratch-resistant": ["scratch-resistant", "scratch resistant"],
        "mirror": ["mirror coating"],
    }


    def extract_optical_suggestions(encounter: Any) -> dict[str, Any]:
        """Pure regex/keyword scan over saved AI Scribe output.

        Returns:
          {"suggestions": [{field, value, matched}, ...], "rationale": str}
        """
        if encounter is None:
            return {"suggestions": [], "rationale": "Walk-in order — no encounter context"}

        ai_text = getattr(encounter, "ai_summary_text", None) or ""
        ap_text = getattr(encounter, "assessment_and_plan", None) or ""
        haystack = (ai_text + " " + ap_text).lower()

        if not haystack.strip():
            return {"suggestions": [], "rationale": "No AI Scribe data on encounter"}

        suggestions: list[dict[str, Any]] = []

        # Lens type — first match wins (priority order matches dict iteration)
        for value, keywords in LENS_TYPE_KEYWORDS.items():
            matched = [kw for kw in keywords if kw.lower() in haystack]
            if matched:
                suggestions.append({"field": "lens_type", "value": value, "matched": matched})
                break

        # Material — first match wins
        for value, keywords in MATERIAL_KEYWORDS.items():
            matched = [kw for kw in keywords if kw.lower() in haystack]
            if matched:
                suggestions.append({"field": "material", "value": value, "matched": matched})
                break

        # Coatings — aggregate all matches
        matched_coatings: list[str] = []
        coating_matches: list[str] = []
        for value, keywords in COATING_KEYWORDS.items():
            matched = [kw for kw in keywords if kw.lower() in haystack]
            if matched:
                matched_coatings.append(value)
                coating_matches.extend(matched)
        if matched_coatings:
            suggestions.append({"field": "coatings", "value": matched_coatings, "matched": coating_matches})

        return {
            "suggestions": suggestions,
            "rationale": "Derived from saved SOAP narrative + Assessment & Plan keywords",
        }
    ```

    No async. No DB. No logging. Pure deterministic input/output.

       ```python
       @router.get("/{order_id}/suggestions/", response_model=OpticalSuggestionsListResponse)
       async def get_optical_suggestions(
           order_id: uuid.UUID,
           ctx: TenantContext = Depends(resolve_tenant_context),
           db: AsyncSession = Depends(get_db_session),
       ) -> dict:
           stmt = select(OpticalOrder).where(OpticalOrder.id == order_id).options(
               selectinload(OpticalOrder.encounter)  # if relationship exists; otherwise db.get
           )
           order = (await db.execute(stmt)).scalar_one_or_none()
           if not order or order.tenant_id != ctx.tenant_id:
               raise HTTPException(404)
           encounter = None
           if order.encounter_id:
               encounter = await db.get(Encounter, order.encounter_id)
           result = extract_optical_suggestions(encounter)
           # Filter out already-resolved
           resolutions = order.suggestion_resolutions_jsonb or {}
           result["suggestions"] = [
               s for s in result["suggestions"]
               if resolutions.get(s["field"]) not in {"accepted", "dismissed"}
           ]
           return result

       _VALID_SUGGESTION_FIELDS = {"lens_type", "material", "coatings"}

       async def _resolve_suggestion(
           order_id: uuid.UUID, field_name: str, outcome: str,
           ctx: TenantContext, db: AsyncSession,
       ) -> OpticalOrder:
           if field_name not in _VALID_SUGGESTION_FIELDS:
               raise HTTPException(400, detail={"error": "invalid_field", "field": field_name})
           order = await db.get(OpticalOrder, order_id)
           if not order or order.tenant_id != ctx.tenant_id:
               raise HTTPException(404)
           if order.status != "draft":
               raise HTTPException(409, detail={"error": "not_draft"})
           staff = await resolve_staff(ctx, db)
           current = dict(order.suggestion_resolutions_jsonb or {})
           current[field_name] = outcome
           order.suggestion_resolutions_jsonb = current
           # Mark JSONB as dirty for SQLAlchemy
           from sqlalchemy.orm.attributes import flag_modified
           flag_modified(order, "suggestion_resolutions_jsonb")
           await log_action(
               db, tenant_id=ctx.tenant_id, action=AuditAction.OPTICAL_ORDER_CONFIGURE_UPDATE,
               actor_id=staff.id if staff else None,
               target_type="optical_order", target_id=order.id,
               metadata={"action": "suggestion_resolution", "field": field_name, "outcome": outcome},
           )
           await db.commit()
           return order

       @router.post("/{order_id}/suggestions/{field_name}/accept/", response_model=OpticalOrderResponse,
                    dependencies=[Depends(require_clinical_action(ClinicalAction.CREATE_OPTICAL_ORDER))])
       async def accept_optical_suggestion(
           order_id: uuid.UUID, field_name: str,
           ctx=Depends(resolve_tenant_context), db=Depends(get_db_session),
       ):
           return await _resolve_suggestion(order_id, field_name, "accepted", ctx, db)

       @router.post("/{order_id}/suggestions/{field_name}/dismiss/", response_model=OpticalOrderResponse,
                    dependencies=[Depends(require_clinical_action(ClinicalAction.CREATE_OPTICAL_ORDER))])
       async def dismiss_optical_suggestion(
           order_id: uuid.UUID, field_name: str,
           ctx=Depends(resolve_tenant_context), db=Depends(get_db_session),
       ):
           return await _resolve_suggestion(order_id, field_name, "dismissed", ctx, db)
       ```

    Critical: use `flag_modified()` after mutating JSONB dict in-place — SQLAlchemy doesn't auto-detect dict mutation, and the audit row would write but the column wouldn't persist without this.
