       import { useLensCatalogStore } from "@/store/lensCatalogStore";

       import { ConfiguratorFooter } from "@/components/optical/configurator/ConfiguratorFooter";

       export default function OpticalOrderConfiguratorPage() {
         const { orderId } = useParams<{ orderId: string; tenant: string }>();
         const router = useRouter();
         const { draft, load, loadSuggestions, flush, suggestions } = useOpticalOrderConfigStore();
         const { loadAll: loadCatalogs } = useLensCatalogStore();
         const [fieldErrors, setFieldErrors] = useState<Array<{ path: string; code: string; message: string }>>([]);

         useEffect(() => {
           if (orderId) {
             load(orderId).then(() => loadSuggestions());
           }
           loadCatalogs();
         }, [orderId]);

         if (!draft) {
           return <div className="p-6 text-[var(--text-secondary)]">Loading configurator…</div>;
         }

         const handleBlur = () => flush();

         return (
           <div className="min-h-screen bg-[var(--bg-glass)] text-[var(--text-primary)]"
                onBlurCapture={handleBlur}>
             <header className="border-b border-[var(--glass-border)] p-4">
               <h1 className="text-lg font-semibold">Configure Order — {draft.status}</h1>
             </header>
             <div className="grid grid-cols-1 lg:grid-cols-2 gap-6 p-6">
               <div>
                 <RxSideBySidePanel
                   habitual={draft.habitualRefraction ?? null}
                   final={draft.finalRefraction ?? null}
                 />
               </div>
               <div className="space-y-6">
                 <FramePicker orderId={draft.id} fieldErrors={fieldErrors} />
                 <LensConfigSection orderId={draft.id} suggestions={suggestions} fieldErrors={fieldErrors} />
                 <MeasurementsSection fitting={draft.fitting} fieldErrors={fieldErrors} />
                 <VisionPlanSection visionPlan={draft.visionPlan} />
               </div>
             </div>
             <ConfiguratorFooter
               status={draft.status}
               onPlace={async () => {
                 setFieldErrors([]);
                 try {
                   await flush();
                   const resp = await fetch(`/api/optical-orders/${draft.id}/place/`, { method: "POST" });
                   if (resp.status === 400) {
                     const detail = (await resp.json()) as { detail?: { field_errors?: Array<{path:string;code:string;message:string}> } };
                     if (detail.detail?.field_errors) setFieldErrors(detail.detail.field_errors);
                     return;
                   }
                   if (!resp.ok) throw new Error(`place ${resp.status}`);
                   await load(draft.id);
                 } catch (e) { console.error(e); }
               }}
               onGenerateJobTicket={async () => {
                 if (draft.status !== "placed") return;
                 const resp = await fetch(`/api/optical-orders/${draft.id}/job-ticket/`, { method: "POST" });
                 if (!resp.ok) return;
                 const blob = await resp.blob();
                 const url = URL.createObjectURL(blob);
                 const a = document.createElement("a");
                 a.href = url;
                 a.download = `job-ticket-${draft.id.slice(0,8)}.pdf`;
                 a.click();
                 URL.revokeObjectURL(url);
                 await load(draft.id);
               }}
               onCancel={() => router.back()}
             />
           </div>
         );
       }

    2. Create `components/optical/configurator/RxSideBySidePanel.tsx`:

       ```tsx
       "use client";
       import { useMemo } from "react";

       interface RxPanelProps {
         habitual: any | null;
         final: any | null;
       }

       const DELTA_THRESHOLD_SE = 0.5;

       function se(rx: any, eye: "od" | "os"): number | null {
         const s = parseFloat(rx?.[`${eye}_sphere`] ?? rx?.[`${eye}Sphere`]);
         const c = parseFloat(rx?.[`${eye}_cylinder`] ?? rx?.[`${eye}Cylinder`]);
         if (Number.isNaN(s)) return null;
         return s + (Number.isNaN(c) ? 0 : c / 2);
       }

       export function RxSideBySidePanel({ habitual, final }: RxPanelProps) {
         const flagged = useMemo(() => {
           const f: ("od" | "os")[] = [];
           for (const eye of ["od", "os"] as const) {
             const h = se(habitual, eye);
             const fi = se(final, eye);
             if (h !== null && fi !== null && Math.abs(fi - h) > DELTA_THRESHOLD_SE) f.push(eye);
           }
           return f;
         }, [habitual, final]);

         if (!habitual && !final) {
           return (
             <div className="rounded border border-[var(--glass-border)] p-4 text-[var(--text-secondary)]">
               No refraction data — PD not captured. Measure with pupillometer.
             </div>
           );
         }

         return (
           <section className="rounded border border-[var(--glass-border)] bg-[var(--bg-glass)] p-4">
             <h2 className="text-sm font-semibold text-[var(--text-primary)] mb-3">Refraction (Habitual | Final)</h2>
             <div className="grid grid-cols-2 gap-4 text-sm">
               <RxColumn label="Habitual" rx={habitual} />
               <RxColumn label="Final" rx={final} />
             </div>
             {flagged.length > 0 && (
               <div className="mt-3 rounded bg-amber-500/10 px-3 py-2 text-xs text-[var(--text-secondary)]">
                 Significant Rx change detected: {flagged.map((e) => e.toUpperCase()).join(", ")} (|Δ SE| &gt; 0.50D)
               </div>
             )}
           </section>
         );
       }

       function RxColumn({ label, rx }: { label: string; rx: any | null }) {
         return (
           <div>
             <div className="text-xs uppercase text-[var(--text-muted)] mb-1">{label}</div>
             <table className="w-full text-xs">
               <thead className="text-[var(--text-muted)]">
                 <tr><th>Eye</th><th>Sph</th><th>Cyl</th><th>Axis</th><th>Add</th></tr>
               </thead>
               <tbody className="text-[var(--text-primary)]">
                 {(["od", "os"] as const).map((eye) => (
                   <tr key={eye}>
                     <td className="font-semibold">{eye.toUpperCase()}</td>
                     <td>{rx?.[`${eye}_sphere`] ?? rx?.[`${eye}Sphere`] ?? "—"}</td>
                     <td>{rx?.[`${eye}_cylinder`] ?? rx?.[`${eye}Cylinder`] ?? "—"}</td>
                     <td>{rx?.[`${eye}_axis`] ?? rx?.[`${eye}Axis`] ?? "—"}</td>
                     <td>{rx?.[`${eye}_add`] ?? rx?.[`${eye}Add`] ?? "—"}</td>
                   </tr>
                 ))}
               </tbody>
             </table>
           </div>
         );
       }
       ```

       Note: the `rx?.[`${eye}_sphere`] ?? rx?.[`${eye}Sphere`]` pattern handles both snake_case (from server response_model_by_alias=False path) AND camelCase (from CamelCaseModel response). Defensive — refraction may use either depending on how it's loaded.

    import { useLensCatalogStore } from "@/store/lensCatalogStore";
    import { useOpticalOrderConfigStore } from "@/store/opticalOrderConfigStore";
    import { SuggestionChip } from "./SuggestionChip";

    export function LensConfigSection({ orderId, suggestions, fieldErrors }: { orderId: string; suggestions: any[]; fieldErrors: any[] }) {
      const { lensTypes, lensMaterials, lensCoatings } = useLensCatalogStore();
      const { draft, patchLineItemLensConfig } = useOpticalOrderConfigStore();
      const lensLine = (draft?.lineItems ?? []).find((li:any) => li.lensConfig != null) ?? (draft?.lineItems ?? [])[0];
      const lc = lensLine?.lensConfig ?? {};

      const lensTypeSuggestion = suggestions.find((s) => s.field === "lens_type");
      const materialSuggestion = suggestions.find((s) => s.field === "material");
      const coatingSuggestion = suggestions.find((s) => s.field === "coatings");

      function setField(key: string, value: any) {
        if (!lensLine) return;
        patchLineItemLensConfig(lensLine.id, { ...lc, [key]: value });
      }

      return (<section className="rounded border border-[var(--glass-border)] bg-[var(--bg-glass)] p-4">
        <h2 className="text-sm font-semibold mb-3 text-[var(--text-primary)]">Lens Configuration</h2>
        <div className="space-y-3">
          <Row label="Type" suggestion={lensTypeSuggestion} onAccept={(val) => {
            const t = lensTypes.find((x) => x.name.toLowerCase().includes(val));
            if (t) setField("lens_type_id", t.id);
          }}>
            <select value={lc.lens_type_id ?? ""} onChange={(e) => setField("lens_type_id", e.target.value)}
              onBlur={() => useOpticalOrderConfigStore.getState().flush()}
              className="w-full px-3 py-2 rounded border border-[var(--glass-border)] bg-transparent text-[var(--text-primary)]">
              <option value="">Select…</option>
              {lensTypes.map((t) => <option key={t.id} value={t.id}>{t.name}</option>)}
            </select>
          </Row>
          <Row label="Material" suggestion={materialSuggestion} onAccept={(val) => {
            const m = lensMaterials.find((x) => x.name.toLowerCase().includes(val));
            if (m) setField("material_id", m.id);
          }}>
            <select value={lc.material_id ?? ""} onChange={(e) => setField("material_id", e.target.value)}
              onBlur={() => useOpticalOrderConfigStore.getState().flush()}
              className="w-full px-3 py-2 rounded border border-[var(--glass-border)] bg-transparent text-[var(--text-primary)]">
              <option value="">Select…</option>
              {lensMaterials.map((m) => <option key={m.id} value={m.id}>{m.name}</option>)}
            </select>
          </Row>
          <Row label="Coatings" suggestion={coatingSuggestion} onAccept={(val: any) => {
            const ids = (Array.isArray(val) ? val : []).map((n: string) => lensCoatings.find((c) => c.name.toLowerCase().includes(n.toLowerCase()))?.id).filter(Boolean);
            setField("coating_ids", ids);
          }}>
            <div className="flex flex-wrap gap-2">
              {lensCoatings.map((c) => {
                const checked = (lc.coating_ids ?? []).includes(c.id);
                return (
                  <label key={c.id} className="text-xs flex items-center gap-1 text-[var(--text-secondary)]">
