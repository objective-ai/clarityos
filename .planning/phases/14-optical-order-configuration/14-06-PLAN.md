    def _seed_lens_reference(session: Session) -> None:
        from backend.db.models.tenant.clinical import LensType, LensMaterial, LensCoating
        from decimal import Decimal
        from sqlalchemy import select as _select

        step("Seeding Lens Reference Catalog")

        LENS_TYPES = [
            ("Single Vision", False, False, 0),
            ("Bifocal", True, False, 1),
            ("Progressive", True, True, 2),
            ("Reading", False, False, 3),
        ]
        for (name, seg, vert, order) in LENS_TYPES:
            existing = session.execute(_select(LensType).where(
                LensType.tenant_id == TENANT_ID,
                LensType.name == name,
                LensType.is_active.is_(True),
            )).scalar_one_or_none()
            if existing is not None:
                continue
            session.add(LensType(
                tenant_id=TENANT_ID, name=name,
                requires_seg_height=seg, requires_vertex=vert, display_order=order,
            ))

        LENS_MATERIALS = [
            ("CR-39", Decimal("1.50"), 58),
            ("Polycarbonate", Decimal("1.59"), 30),
            ("Trivex", Decimal("1.53"), 43),
            ("Hi-Index 1.67", Decimal("1.67"), 32),
            ("Hi-Index 1.74", Decimal("1.74"), 33),
            ("Hi-Index 1.80", Decimal("1.80"), 30),
        ]
        for (name, ri, abbe) in LENS_MATERIALS:
            existing = session.execute(_select(LensMaterial).where(
                LensMaterial.tenant_id == TENANT_ID,
                LensMaterial.name == name,
                LensMaterial.is_active.is_(True),
            )).scalar_one_or_none()
            if existing is not None:
                continue
            session.add(LensMaterial(
                tenant_id=TENANT_ID, name=name,
                refractive_index=ri, abbe_value=abbe,
            ))

        LENS_COATINGS = [
            ("AR", "treatment"),
            ("UV", "treatment"),
            ("Blue Light", "treatment"),
            ("Photochromic", "treatment"),
            ("Polarized", "tint"),
            ("Scratch-Resistant", "finish"),
            ("Mirror", "finish"),
        ]
        for (name, category) in LENS_COATINGS:
            existing = session.execute(_select(LensCoating).where(
                LensCoating.tenant_id == TENANT_ID,
                LensCoating.name == name,
                LensCoating.is_active.is_(True),
            )).scalar_one_or_none()
            if existing is not None:
                continue
            session.add(LensCoating(
                tenant_id=TENANT_ID, name=name, category=category,
            ))

        session.flush()
