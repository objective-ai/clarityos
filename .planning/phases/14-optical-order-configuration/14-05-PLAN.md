    def _fmt_decimal(v: Any, fallback: str = "—") -> str:
        if v is None:
            return fallback
        try:
            return f"{Decimal(str(v)):+.2f}"
        except Exception:
            return str(v)


    def _fmt_int(v: Any, fallback: str = "—") -> str:
        if v is None:
            return fallback
        try:
            return str(int(v))
        except Exception:
            return str(v)


    def _settings(tenant: Any, key: str, fallback: str = "—") -> str:

    def _section_style() -> TableStyle:
        return TableStyle([
            ("FONTNAME", (0, 0), (-1, -1), "Courier"),
            ("FONTSIZE", (0, 0), (-1, -1), 9),
            ("BOX", (0, 0), (-1, -1), 0.5, colors.black),
            ("INNERGRID", (0, 0), (-1, -1), 0.25, colors.grey),
            ("VALIGN", (0, 0), (-1, -1), "MIDDLE"),
            ("LEFTPADDING", (0, 0), (-1, -1), 4),

        ])


    def _build_rx_table_data(habitual: Any, final: Any) -> list[list[str]]:
        def row(eye: str, rx: Any) -> list[str]:
            if rx is None:
                return [eye, "—", "—", "—", "—"]
            sphere = _fmt_decimal(getattr(rx, f"{eye.lower()}_sphere", None))
            cyl = _fmt_decimal(getattr(rx, f"{eye.lower()}_cylinder", None))
            axis = _fmt_int(getattr(rx, f"{eye.lower()}_axis", None))
            add = _fmt_decimal(getattr(rx, f"{eye.lower()}_add", None))
            return [eye, sphere, cyl, axis, add]
        # Two-column layout: [Eye, Sph, Cyl, Axis, Add] | [Eye, Sph, Cyl, Axis, Add]
        return [
            ["", "Habitual", "", "", "", "", "Final", "", "", ""],
            ["Eye", "Sph", "Cyl", "Axis", "Add", "Eye", "Sph", "Cyl", "Axis", "Add"],
            row("OD", habitual) + row("OD", final),
            row("OS", habitual) + row("OS", final),
        ]

    from datetime import datetime, timezone
    from fastapi import Response as FastAPIResponse
    from backend.services.job_ticket_pdf import build_job_ticket_pdf
    from backend.db.models.tenant.clinical import LensType, LensMaterial, LensCoating

    @router.post("/{order_id}/job-ticket/",
                 dependencies=[Depends(require_clinical_action(ClinicalAction.GENERATE_JOB_TICKET))])
    async def generate_job_ticket(
        order_id: uuid.UUID,
        ctx: TenantContext = Depends(resolve_tenant_context),
        db: AsyncSession = Depends(get_db_session),
    ) -> FastAPIResponse:
        stmt = (
            select(OpticalOrder)
            .where(OpticalOrder.id == order_id)
            .options(
                selectinload(OpticalOrder.line_items),
                selectinload(OpticalOrder.final_refraction),
                selectinload(OpticalOrder.habitual_refraction),
                selectinload(OpticalOrder.patient),  # if relationship exists
            )
        )
        order = (await db.execute(stmt)).scalar_one_or_none()
        if not order or order.tenant_id != ctx.tenant_id:
            raise HTTPException(404)
        if order.status != "placed":
            raise HTTPException(400, detail={"error": "not_placed",
                                              "message": "Place order before generating job ticket"})

        # Resolve patient + encounter + tenant
        from backend.db.models.tenant.clinical import Patient, Encounter
        patient = await db.get(Patient, order.patient_id)
        encounter = await db.get(Encounter, order.encounter_id) if order.encounter_id else None
        from backend.db.models.public.saas import Tenant
        tenant = await db.get(Tenant, ctx.tenant_id)

        # Batch lookup all products + lens refs referenced by line items
        product_ids = {li.product_id for li in order.line_items}
        products = (await db.execute(select(Product).where(Product.id.in_(product_ids)))).scalars().all()
        products_by_id = {p.id: p for p in products}

        lens_type_ids: set = set()
        lens_material_ids: set = set()
        lens_coating_ids: set = set()
        for li in order.line_items:
            lc = li.lens_config_jsonb or {}
            if lc.get("lens_type_id"):
                lens_type_ids.add(uuid.UUID(lc["lens_type_id"]) if isinstance(lc["lens_type_id"], str) else lc["lens_type_id"])
            if lc.get("material_id"):
                lens_material_ids.add(uuid.UUID(lc["material_id"]) if isinstance(lc["material_id"], str) else lc["material_id"])
            for cid in lc.get("coating_ids", []):
                lens_coating_ids.add(uuid.UUID(cid) if isinstance(cid, str) else cid)

        lens_types_by_id = {}
        if lens_type_ids:
            rows = (await db.execute(select(LensType).where(LensType.id.in_(lens_type_ids)))).scalars().all()
            lens_types_by_id = {r.id: r for r in rows}

        lens_materials_by_id = {}
        if lens_material_ids:
            rows = (await db.execute(select(LensMaterial).where(LensMaterial.id.in_(lens_material_ids)))).scalars().all()
            lens_materials_by_id = {r.id: r for r in rows}

        lens_coatings_by_id = {}
        if lens_coating_ids:
            rows = (await db.execute(select(LensCoating).where(LensCoating.id.in_(lens_coating_ids)))).scalars().all()
            lens_coatings_by_id = {r.id: r for r in rows}

        staff = await resolve_staff(ctx, db)
        staff_name = getattr(staff, "full_name", None) or "—"

        pdf_bytes = build_job_ticket_pdf(
            order=order, patient=patient, encounter=encounter,
            final_refraction=order.final_refraction, habitual_refraction=order.habitual_refraction,
            products_by_id=products_by_id,
            lens_types_by_id=lens_types_by_id,
            lens_materials_by_id=lens_materials_by_id,
            lens_coatings_by_id=lens_coatings_by_id,
            tenant=tenant, staff_name=staff_name,
        )

        # Persist generation timestamp + audit in primary TXN
        order.job_ticket_generated_at = datetime.now(timezone.utc)
        await log_action(
            db, tenant_id=ctx.tenant_id, action=AuditAction.JOB_TICKET_GENERATE,
            actor_id=staff.id if staff else None,
            target_type="optical_order", target_id=order.id,
            metadata={"order_id": str(order.id), "byte_count": len(pdf_bytes)},
        )
        await db.commit()

        filename = f"job-ticket-{str(order.id)[:8]}.pdf"
        return FastAPIResponse(
            content=pdf_bytes,
            media_type="application/pdf",
            headers={"Content-Disposition": f'attachment; filename="{filename}"'},
        )
