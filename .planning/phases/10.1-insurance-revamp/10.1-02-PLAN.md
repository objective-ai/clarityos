**EligibilityBadge component** (inline in InsuranceTab.tsx or extracted):
```tsx
const ELIGIBILITY_COLORS: Record<string, string> = {
  active: "bg-emerald-400",
  inactive: "bg-red-400",
  pending_verification: "bg-yellow-400",
  expired: "bg-orange-400",
  unknown: "bg-gray-400",
};

const ELIGIBILITY_LABELS: Record<string, string> = {
  active: "Active",
  inactive: "Inactive",
  pending_verification: "Pending",
  expired: "Expired",
  unknown: "Unknown",
};

function EligibilityBadge({ status, verifiedDate }: { status: string; verifiedDate: string | null }) {
  const isStale = verifiedDate
    ? (Date.now() - new Date(verifiedDate).getTime()) > 30 * 24 * 60 * 60 * 1000
    : false;
  return (
    <div className="flex items-center gap-1.5">
      <span className={`w-2 h-2 rounded-full ${ELIGIBILITY_COLORS[status] ?? "bg-gray-400"}`} />
      <span className="text-[13px] font-medium">{ELIGIBILITY_LABELS[status] ?? status}</span>
      {verifiedDate && (
        <span className={`text-[11px] ${isStale ? "text-orange-400" : "text-[var(--text-muted)]"}`}>
          {isStale && "⚠ "}verified {verifiedDate}
        </span>
      )}
    </div>
  );
}
```
